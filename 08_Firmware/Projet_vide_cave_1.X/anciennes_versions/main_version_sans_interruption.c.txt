/*
 * Projet vide-cave - IUT Lyon 1 GEII
 *
 * Test avec le potentiometre sur RA0, il remplace le capteur de courant :
 *   potentiometre en bas    : courant faible -> pompe a vide
 *   potentiometre au milieu : courant normal -> marche
 *   potentiometre en haut   : courant fort   -> moteur bloque
 *
 * La LED sur RB0 montre si le moteur est commande, sans avoir de 230 V.
 */

#include <xc.h>
#include "mcc_generated_files/system/system.h"
#include "lcd.h"

// Seuils de courant en points du CAN (de 0 a 255)
#define SEUIL_BAS   91
#define SEUIL_HAUT  173

// Durees en tours de boucle (1 tour = 100 ms)
#define DUREE_ARRET      20    // 20 x 100 ms = 2 s
#define DUREE_DEMARRAGE  50    // 50 x 100 ms = 5 s
#define DUREE_DEFAUT     100   // 100 x 100 ms = 10 s

// Les etats du programme
#define ARRET      0
#define DEMARRAGE  1
#define MARCHE     2
#define BLOQUE     3
#define A_VIDE     4

// Prototypes des fonctions
void Moteur_Marche(void);
void Moteur_Arret(void);
uint8_t Lire_Courant(void);
void Afficher_Etat(uint8_t etat_moteur);

void main(void)
{
    uint8_t etat = ARRET;
    uint8_t etat_affiche = ARRET;
    uint8_t compteur = 0;
    uint8_t courant;

    SYSTEM_Initialize();      // MCC : horloge, broches et CAN
    LCD_Initialize();         // toujours apres SYSTEM_Initialize
    Moteur_Arret();           // securite : moteur coupe au demarrage
    Afficher_Etat(etat);

    while(1) // ne se termine jamais
    {
        courant = Lire_Courant();

        switch(etat)
        {
            case ARRET:
                compteur = compteur + 1;
                if(compteur >= DUREE_ARRET)      // pour le test : on demarre apres 2 s
                {
                    compteur = 0;
                    Moteur_Marche();
                    etat = DEMARRAGE;
                }
                break;

            case DEMARRAGE:
                compteur = compteur + 1;         // on laisse passer le pic de courant
                if(compteur >= DUREE_DEMARRAGE)
                {
                    compteur = 0;
                    if(courant > SEUIL_HAUT)     // encore trop fort apres 5 s
                    {
                        Moteur_Arret();
                        etat = BLOQUE;
                    }
                    else
                    {
                        etat = MARCHE;
                    }
                }
                break;

            case MARCHE:
                if(courant > SEUIL_HAUT)         // courant trop fort
                {
                    compteur = compteur + 1;
                    if(compteur >= DUREE_DEFAUT)
                    {
                        Moteur_Arret();
                        etat = BLOQUE;
                    }
                }
                else if(courant < SEUIL_BAS)     // courant trop faible
                {
                    compteur = compteur + 1;
                    if(compteur >= DUREE_DEFAUT)
                    {
                        Moteur_Arret();
                        etat = A_VIDE;
                    }
                }
                else                             // courant normal
                {
                    compteur = 0;
                }
                break;

            case BLOQUE:                         // on n'en sort que par le bouton reset
                break;

            case A_VIDE:                         // on n'en sort que par le bouton reset
                break;
        }

        if(etat != etat_affiche)                 // on affiche seulement si l'etat change
        {
            Afficher_Etat(etat);
            etat_affiche = etat;
        }

        __delay_ms(100);
    }
}

void Moteur_Marche(void)
{
    OPTO_TRIAC_SetHigh();     // RB2 a 1 : l'optotriac declenche le triac
    LED_SetHigh();            // RB0 a 1 : la LED s'allume
}

void Moteur_Arret(void)
{
    OPTO_TRIAC_SetLow();
    LED_SetLow();
}

// CAN sur 12 bits, resultat range a gauche sur 16 bits.
// On garde les 8 bits de gauche : valeur de 0 a 255.
// Quantum : q = 5 V / 256 = 19,5 mV
uint8_t Lire_Courant(void)
{
    uint16_t mesure;

    mesure = ADC_GetConversion(Capteur_COURANT, negChannel_ADNREF);
    return (uint8_t)(mesure >> 8);
}

void Afficher_Etat(uint8_t etat_moteur)
{
    LCD_Clear();
    LCD_GoTo(1, 1);
    LCD_WriteText("Etat moteur :");
    LCD_GoTo(2, 1);

    switch(etat_moteur)
    {
        case ARRET:
            LCD_WriteText("Arret");
            break;
        case DEMARRAGE:
            LCD_WriteText("Demarrage");
            break;
        case MARCHE:
            LCD_WriteText("En marche");
            break;
        case BLOQUE:
            LCD_WriteText("BLOQUE - coupe");
            break;
        case A_VIDE:
            LCD_WriteText("A VIDE - coupe");
            break;
    }
}
